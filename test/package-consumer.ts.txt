import {
  accumulate,
  renderTimeline,
  rota,
  schedule,
  tally,
  weekdays,
} from "@kensio/quando";
import { explainRule, layer, merged } from "@kensio/quando/core";
import { parseSchedule } from "@kensio/quando/parsing";

interface Duty {
  readonly person: string;
  readonly level: number;
}

const office = schedule({ zone: "Europe/London" }).open(
  weekdays(),
  "09:00-17:00",
  { label: "Regular office hours" },
);
const staffing = merged("sum", layer(weekdays(), 2));
const staffHours = tally()
  .plus(weekdays(), 2)
  .totalBetween(
    Temporal.ZonedDateTime.from("2026-03-09T00:00[Europe/London]"),
    Temporal.ZonedDateTime.from("2026-03-10T00:00[Europe/London]"),
    "hour",
  );
const weekdayStaff = tally()
  .plus(weekdays(), 2)
  .countAt(Temporal.ZonedDateTime.from("2026-03-09T11:00[Europe/London]"));
const accumulated = accumulate(
  staffing,
  {
    from: Temporal.ZonedDateTime.from("2026-03-09T00:00[Europe/London]"),
    to: Temporal.ZonedDateTime.from("2026-03-10T00:00[Europe/London]"),
  },
  "hour",
);
const timeline = office.renderTimeline(
  Temporal.ZonedDateTime.from("2026-03-09T00:00[Europe/London]"),
  Temporal.ZonedDateTime.from("2026-03-10T00:00[Europe/London]"),
);
const standaloneTimeline = renderTimeline(weekdays(), {
  from: Temporal.ZonedDateTime.from("2026-03-09T00:00[Europe/London]"),
  to: Temporal.ZonedDateTime.from("2026-03-10T00:00[Europe/London]"),
});
const rotaWithInterface = rota<Duty>().assign(weekdays(), {
  person: "alice",
  level: 2,
});
const weekdayExplanation = explainRule(
  weekdays(),
  Temporal.ZonedDateTime.from("2026-03-09T10:00[Europe/London]"),
);

// @ts-expect-error The sum strategy only accepts numeric layer values.
merged("sum", layer(weekdays(), "two"));

void [
  office,
  staffing,
  staffHours,
  weekdayStaff,
  accumulated,
  timeline,
  standaloneTimeline,
  rotaWithInterface,
  weekdayExplanation,
  parseSchedule,
];
